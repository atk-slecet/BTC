#   + 全国 margin * 广东全国风向敏感度
#
# 正数 = 民主宪政联盟领先
# 负数 = 人民民主统一战线领先
# =========================================================
CHI_calculate_guangdong_margin = {

    # -----------------------------------------
    # 1. 计算全国蓝红差
    # -----------------------------------------

    set_variable = {
        CHI_national_margin = CHI_DCA_support
    }

    subtract_from_variable = {
        CHI_national_margin = CHI_PDUF_support
    }


    # -----------------------------------------
    # 2. 复制全国 margin 给广东临时计算变量
    # -----------------------------------------

    set_variable = {
        CHI_GD_national_effect = CHI_national_margin
    }


    # -----------------------------------------
    # 3. 乘以广东对全国风向的敏感度
    # -----------------------------------------

    multiply_variable = {
        CHI_GD_national_effect = CHI_GD_national_swing_weight
    }


    # -----------------------------------------
    # 4. 广东 margin 从地方基本盘开始
    # -----------------------------------------

    set_variable = {
        CHI_GD_margin = CHI_GD_regional_bias
    }


    # -----------------------------------------
    # 5. 加入全国政治风向
    # -----------------------------------------

    add_to_variable = {
        CHI_GD_margin = CHI_GD_national_effect
    }


    # -----------------------------------------
    # 6. 根据 margin 分类
    # -----------------------------------------

    CHI_classify_guangdong_result = yes
}


CHI_classify_guangdong_result = {

    set_variable = {
        CHI_GD_map_state = 2
    }

    # =========================
    # 先判断政治状态
    # =========================

    if = {
        limit = {
            NOT = {
                check_variable = {
                    CHI_GD_margin < 10
                }
            }
        }

        set_variable = {
            CHI_GD_map_state = 1
        }
    }

    else_if = {
        limit = {
            NOT = {
                check_variable = {
                    CHI_GD_margin > -10
                }
            }
        }

        set_variable = {
            CHI_GD_map_state = 3
        }
    }


    # =========================
    # 再转换为地图 Frame
    # =========================

    if = {
        limit = {
            check_variable = {
                CHI_GD_map_state = 1
            }
        }

        set_variable = {
            CHI_guangdong_map_frame = 2
        }
    }

    else_if = {
        limit = {
            check_variable = {
                CHI_GD_map_state = 2
            }
        }

        set_variable = {
            CHI_guangdong_map_frame = 4
        }
    }

    else_if = {
        limit = {
            check_variable = {
                CHI_GD_map_state = 3
            }
        }

        set_variable = {
            CHI_guangdong_map_frame = 3
        }
    }
}

    # 默认摇摆
    set_variable = {
        CHI_REG_state = 2
    }

    # 蓝

        limit = {
            NOT = {
                check_variable = {
                    CHI_REG_margin < 10
                }
            }
        }

        set_variable = {
            CHI_REG_state = 1
        }
    }

    # 红
    else_if = {
        limit = {
            NOT = {
                check_variable = {
                    CHI_REG_margin > -10
                }
            }
        }

        set_variable = {
            CHI_REG_state = 3
        }
    }
}

# =========================================================
# 08C-4
# 山西地图显示映射
#
# State:
# 1 = 蓝
# 2 = 摇摆
# 3 = 红
#
# DDS:
# Frame 1 = 白
# Frame 2 = 蓝
# Frame 3 = 红
# Frame 4 = 摇摆
# =========================================================

CHI_update_shanxi_map_frame = {

    if = {
        limit = {
            check_variable = {
                CHI_SX_map_state = 1
            }
        }

        set_variable = {
            CHI_shanxi_map_frame = 2
        }
    }

    else_if = {
        limit = {
            check_variable = {
                CHI_SX_map_state = 2
            }
        }

        set_variable = {
            CHI_shanxi_map_frame = 4
        }
    }

    else_if = {
        limit = {
            check_variable = {
                CHI_SX_map_state = 3
            }
        }

        set_variable = {
            CHI_shanxi_map_frame = 3
        }
    }
}
